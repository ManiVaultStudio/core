cmake_minimum_required(VERSION 3.22)

set(POINTDATAVIEWPLUGIN "PointDataViewPlugin")
project(${POINTDATAVIEWPLUGIN} LANGUAGES CXX)

set(CMAKE_INCLUDE_CURRENT_DIR ON)

set(PLUGIN
    src/PointDataViewPlugin.h
    src/PointDataViewPlugin.cpp
)

set(WIDGETS
    src/PointDataViewWidget.h
    src/PointDataViewWidget.cpp
)

set(AUX
    PluginInfo.json
)

set(POINTDATA_VIEW_SOURCES ${PLUGIN} ${WIDGETS} ${AUX})

source_group(Plugin FILES ${PLUGIN})
source_group(Widgets FILES ${WIDGETS})
source_group(Aux FILES ${AUX})

add_library(${POINTDATAVIEWPLUGIN} SHARED ${POINTDATA_VIEW_SOURCES})

target_include_directories(${POINTDATAVIEWPLUGIN} PRIVATE "${MV_INSTALL_DIR}/$<CONFIGURATION>/include/")
target_include_directories(${POINTDATAVIEWPLUGIN} PRIVATE "${MV_INSTALL_DIR}/$<CONFIGURATION>/include/PointData/")
target_compile_features(${POINTDATAVIEWPLUGIN} PRIVATE cxx_std_${MV_CXX_STANDARD})

target_link_libraries(${POINTDATAVIEWPLUGIN} PRIVATE Qt6::Core Qt6::Widgets Qt6::Network Qt6::WebEngineWidgets ${MV_PUBLIC_LIB} PointData)

mv_check_and_set_AVX(${POINTDATAVIEWPLUGIN} ${MV_USE_AVX})

set_target_properties(${POINTDATAVIEWPLUGIN} PROPERTIES
    AUTOMOC ON
    UNITY_BUILD ${MV_UNITY_BUILD}
)

install(TARGETS ${POINTDATAVIEWPLUGIN}
    RUNTIME DESTINATION Plugins COMPONENT PLUGINS
    LIBRARY DESTINATION Plugins COMPONENT PLUGINS
)

add_custom_command(TARGET ${POINTDATAVIEWPLUGIN} POST_BUILD
    COMMAND ${CMAKE_COMMAND}
        --install ${CMAKE_CURRENT_BINARY_DIR}
        --config $<CONFIGURATION>
        --prefix ${MV_INSTALL_DIR}/$<CONFIGURATION>
)

mv_handle_plugin_config(${POINTDATAVIEWPLUGIN})

if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    set_property(TARGET ${POINTDATAVIEWPLUGIN} PROPERTY VS_DEBUGGER_WORKING_DIRECTORY $<IF:$<CONFIG:DEBUG>,${MV_INSTALL_DIR}/Debug,$<IF:$<CONFIG:RELWITHDEBINFO>,${MV_INSTALL_DIR}/RelWithDebInfo,${MV_INSTALL_DIR}/Release>>)
    set_property(TARGET ${POINTDATAVIEWPLUGIN} PROPERTY VS_DEBUGGER_COMMAND $<IF:$<CONFIG:DEBUG>,"${MV_INSTALL_DIR}/Debug/${MV_APPLICATION_NAME}.exe",$<IF:$<CONFIG:RELWITHDEBINFO>,"${MV_INSTALL_DIR}/RelWithDebInfo/${MV_APPLICATION_NAME}.exe","${MV_INSTALL_DIR}/Release/${MV_APPLICATION_NAME}.exe">>)
endif()
